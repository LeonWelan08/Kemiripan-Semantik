# SEARCH
def search_top(df,
               query,
               sim_col="Similarity",
               doc_col="Top_Document"):

    data = df[
        df["Query"].str.contains(
            query,
            case=False,
            na=False
        )
    ]

    if data.empty:
        return None, 0

    top = data.sort_values(
        sim_col,
        ascending=False
    ).iloc[0]

    return top[doc_col], top[sim_col]
